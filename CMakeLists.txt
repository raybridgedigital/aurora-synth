cmake_minimum_required(VERSION 3.25)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0 CACHE STRING "Minimum macOS")
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Architecture")
project(Aurora VERSION 0.23.0 LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(VSTSDK "${CMAKE_SOURCE_DIR}/build/deps/vst3sdk")
# The SDK uses XCODE_VERSION only for compiler/linker feature gates, even
# with Makefiles. Command Line Tools carry the same Apple toolchain version.
execute_process(COMMAND xcodebuild -version RESULT_VARIABLE XCODE_STATUS OUTPUT_QUIET ERROR_QUIET)
if(NOT XCODE_STATUS EQUAL 0)
    execute_process(COMMAND pkgutil --pkg-info com.apple.pkg.CLTools_Executables OUTPUT_VARIABLE CLT_INFO)
    string(REGEX MATCH "version: ([0-9]+)" CLT_MATCH "${CLT_INFO}")
    set(XCODE_VERSION "${CMAKE_MATCH_1}")
    if(NOT XCODE_VERSION)
        message(FATAL_ERROR "Apple Command Line Tools are required")
    endif()
    set(ENV{XCODE_VERSION} "${XCODE_VERSION}")
endif()
add_subdirectory("${VSTSDK}" vst-sdk)
smtg_enable_vst3_sdk()
smtg_add_vst3plugin(Aurora Sources/PluginVST.mm Sources/PluginCore.mm Sources/PluginBridge.mm Sources/SynthEngine.cpp "${VSTSDK}/public.sdk/source/vst/vstsinglecomponenteffect.cpp")
target_include_directories(Aurora PRIVATE "${VSTSDK}" Sources)
set_source_files_properties(Sources/PluginVST.mm Sources/PluginCore.mm Sources/PluginBridge.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
target_link_libraries(Aurora PRIVATE sdk "${CMAKE_SOURCE_DIR}/build/plugin/libAuroraUI.a" "-framework SwiftUI" "-framework AppKit" "-framework Foundation" "-framework AudioToolbox")
target_link_directories(Aurora PRIVATE /Library/Developer/CommandLineTools/usr/lib/swift/macosx)
target_link_options(Aurora PRIVATE "-Wl,-rpath,/usr/lib/swift")
smtg_target_set_bundle(Aurora BUNDLE_IDENTIFIER com.raybridgedigital.aurora.vst3 COMPANY_NAME "Ray Bridge Digital")
set_target_properties(Aurora PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER com.raybridgedigital.aurora.vst3 MACOSX_BUNDLE_BUNDLE_NAME Aurora MACOSX_BUNDLE_BUNDLE_VERSION 0.23.0 MACOSX_BUNDLE_SHORT_VERSION_STRING 0.23.0)
smtg_target_add_plugin_resources(Aurora RESOURCES Resources/Aurora100.json Resources/AuroraPrism100.json Resources/AuroraNova100.json Resources/AuroraReference.json Resources/AuroraGB109.json)
smtg_target_add_plugin_resources(Aurora RESOURCES Resources/PluginLicenses/VST3.txt Resources/PluginLicenses/AudioUnitSDK.txt)
add_custom_command(TARGET Aurora POST_BUILD
    COMMAND /usr/bin/xattr -cr "$<TARGET_BUNDLE_DIR:Aurora>"
    COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:Aurora>")

set(AUSDK "${CMAKE_SOURCE_DIR}/build/deps/AudioUnitSDK")
file(GLOB AU_SOURCES CONFIGURE_DEPENDS "${AUSDK}/src/AudioUnitSDK/*.cpp")
add_library(AuroraAUSDK STATIC ${AU_SOURCES})
set_target_properties(AuroraAUSDK PROPERTIES CXX_STANDARD 23)
target_include_directories(AuroraAUSDK PUBLIC "${AUSDK}/include")
set(AUWRAPPER "${VSTSDK}/public.sdk/source/vst/auwrapper")
file(WRITE "${CMAKE_BINARY_DIR}/aucocoaclassprefix.h" "#define SMTG_AUCocoaUIBase_CLASS_NAME AuroraCocoaViewFactory\n")
add_library(AuroraAU MODULE "${AUWRAPPER}/auwrapper.mm" "${AUWRAPPER}/aucocoaview.mm" "${AUWRAPPER}/NSDataIBStream.mm")
set_target_properties(AuroraAU PROPERTIES CXX_STANDARD 23 OBJCXX_STANDARD 23)
target_include_directories(AuroraAU PRIVATE "${VSTSDK}" "${CMAKE_BINARY_DIR}")
target_compile_definitions(AuroraAU PRIVATE SMTG_AUWRAPPER_USES_AUSDK CA_USE_AUDIO_PLUGIN_ONLY=0)
target_link_libraries(AuroraAU PRIVATE sdk_hosting AuroraAUSDK "-framework AudioUnit" "-framework CoreMIDI" "-framework AudioToolbox" "-framework CoreFoundation" "-framework Carbon" "-framework Cocoa" "-framework CoreAudio")
set_target_properties(AuroraAU PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION component PREFIX "" OUTPUT_NAME Aurora MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/Resources/AuroraAU-Info.plist")
add_dependencies(AuroraAU Aurora)
add_executable(PluginHostChecks Tests/PluginHostChecks.mm "${VSTSDK}/public.sdk/source/common/memorystream.cpp")
target_include_directories(PluginHostChecks PRIVATE "${VSTSDK}")
target_compile_options(PluginHostChecks PRIVATE -UNDEBUG -fobjc-arc)
target_link_libraries(PluginHostChecks PRIVATE sdk_hosting "-framework AppKit")
add_custom_command(TARGET AuroraAU POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:AuroraAU>/Contents/Resources"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "$<TARGET_BUNDLE_DIR:Aurora>" "$<TARGET_BUNDLE_DIR:AuroraAU>/Contents/Resources/plugin.vst3"
    COMMAND /usr/bin/xattr -cr "$<TARGET_BUNDLE_DIR:AuroraAU>"
    COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:AuroraAU>")
